extends RefCounted
const Layout=preload("res://scripts/ui_restore/layout.gd")
const Typography=preload("res://scripts/ui_restore/typography.gd")
const ROOT="res://assets/art_review/settings_guide_entry_v1/unpacked/STATIC_SETTINGS_GUIDE_ENTRY_PATCH_CANDIDATE_v1/"
const NIGHT="res://assets/art_review/night_state_supplement_v1/unpacked/STATIC_NIGHT_STATE_SUPPLEMENT_v1/assets/"
var scenes: Dictionary=JSON.parse_string(FileAccess.get_file_as_string(ROOT+"scenes_and_labels.json"))
var dependencies: Array=JSON.parse_string(FileAccess.get_file_as_string(ROOT+"dependency_files.json"))
var textures: Dictionary={}
var dropdown: Control
var errors: Array=[]
func scene(locale: String, expanded: bool) -> Dictionary:
	return scenes["pause_settings_guide_%s_music_on_sound_on_%s" % [locale,"expanded" if expanded else "closed"]]
func texture(view,uri: String) -> Texture2D:
	if textures.has(uri): return textures[uri]
	var path: String=NIGHT+uri.get_file() if uri.contains("STATIC_NIGHT_STATE_SUPPLEMENT") else "res://assets/ui_restore/modal_normal_v3/"+uri.get_file()
	for dependency in dependencies:
		if dependency.file==uri and FileAccess.get_sha256(path)!=dependency.sha256:
			errors.append("Settings guide dependency hash mismatch: "+path)
			return null
	var pixels:=Image.new()
	if pixels.load_png_from_buffer(FileAccess.get_file_as_bytes(path))!=OK: return null
	textures[uri]=ImageTexture.create_from_image(pixels)
	return textures[uri]
func image_layer(view,parent: Node,layer: Dictionary) -> void:
	var tex=texture(view,layer.file)
	if tex==null: return
	if layer.has("9patch_margins_px"):
		var node:=NinePatchRect.new()
		node.name=layer.id
		node.texture=tex
		node.position=Layout.reference_transform()*Vector2(layer.top_left_px[0],layer.top_left_px[1])
		node.size=Vector2(layer.display_size_px[0],layer.display_size_px[1])
		node.scale=Layout.reference_transform().get_scale()
		node.patch_margin_left=int(layer["9patch_margins_px"][0])
		node.patch_margin_top=int(layer["9patch_margins_px"][1])
		node.patch_margin_right=int(layer["9patch_margins_px"][2])
		node.patch_margin_bottom=int(layer["9patch_margins_px"][3])
		node.mouse_filter=Control.MOUSE_FILTER_IGNORE
		node.z_index=int(layer.get("z",0))
		parent.add_child(node)
	else:
		var node:=Sprite2D.new()
		node.texture=tex
		node.name=layer.get("id",layer.file.get_file())
		node.position=Layout.reference_transform()*(Vector2(layer.top_left_px[0],layer.top_left_px[1])+Vector2(tex.get_size())*0.5)
		node.scale=Layout.reference_transform().get_scale()
		node.rotation_degrees=float(layer.get("rotation_degrees",0))
		node.z_index=int(layer.get("z",0))
		parent.add_child(node)
func label(view,parent: Node,spec: Dictionary) -> void:
	var box: Array=spec.rect_px
	var rect:=Layout.reference_transform()*Rect2(box[0],box[1],box[2],box[3])
	var node=view._label(parent,"GuidePatch_"+spec.id,rect,spec.text,int(float(spec.font_px)*Layout.reference_transform().get_scale().x))
	node.horizontal_alignment=HORIZONTAL_ALIGNMENT_CENTER
	node.add_theme_font_override("font",Typography.font("latin" if spec.get("font_asset","").ends_with(".ttf") else "cjk_face0"))
	node.add_theme_color_override("font_color",Color(spec.color))
	node.z_index=80
func build(view,input: Control) -> void:
	var spec=scene(view.i18n.locale,false)
	for layer in spec.layers:
		if layer.get("id","") in ["pause_panel_extended","how_to_play_button"]: image_layer(view,view.modal_art,layer)
		elif layer.get("id","").begins_with("modal_language_dropdown_") or layer.get("id","")=="language_caret_down":
			var copy=layer.duplicate(true)
			copy.z=45 if copy.id.begins_with("modal_language") else 48
			image_layer(view,view.modal_art,copy)
	for text in spec.labels: label(view,view.modal_texts,text)
	view._button(input,"Modal_how_to_play",Layout.reference_transform()*Rect2(165,1350,610,128),{"type":"tutorial_review"})
func is_open() -> bool: return is_instance_valid(dropdown) and dropdown.visible and not dropdown.is_queued_for_deletion()
func close() -> void:
	if is_instance_valid(dropdown):
		dropdown.hide()
		dropdown.queue_free()
func open(view) -> void:
	if is_open(): close(); return
	dropdown=Control.new()
	dropdown.name="PauseLanguagePopover"
	dropdown.size=Layout.CANVAS
	dropdown.mouse_filter=Control.MOUSE_FILTER_STOP
	dropdown.z_index=200
	view.modal_content.add_child(dropdown)
	dropdown.gui_input.connect(func(event):
		if event is InputEventMouseButton and event.pressed:
			dropdown.accept_event()
			close()
	)
	var spec=scene(view.i18n.locale,true)
	for layer in spec.layers:
		if layer.file.contains("pause_language_expanded") or layer.file.contains("caret_up") or layer.file.contains("selected_check"): image_layer(view,dropdown,layer)
	for text in spec.labels:
		if text.id in ["locale_zh","locale_en"]: label(view,dropdown,text)
	for target in spec.active_hit_targets:
		var box: Array=target.rect
		var button:=Button.new()
		button.name=target.id
		button.position=Layout.reference_transform()*Vector2(box[0],box[1])
		button.size=Vector2(box[2],box[3])*Layout.reference_transform().get_scale()
		button.flat=true
		button.z_index=100
		button.focus_mode=Control.FOCUS_NONE
		for state in ["normal","hover","pressed","disabled","focus"]: button.add_theme_stylebox_override(state,StyleBoxEmpty.new())
		dropdown.add_child(button)
		var locale: String="zh" if target.id=="choose_zh" else "en"
		button.pressed.connect(func():
			close()
			view._send({"type":"language_menu","locale":locale,"epoch":view.snapshot.epoch})
		)
