extends RefCounted

const Layout = preload("res://scripts/ui_restore/layout.gd")
const Typography = preload("res://scripts/ui_restore/typography.gd")
const DynamicText = preload("res://scripts/ui_restore/dynamic_text.gd")
const DIRECTORY := "res://assets/ui_restore/pause_off_v1/"
var spec: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(DIRECTORY + "scene_placement_and_labels.json"))
var textures: Dictionary = {}
var errors: Array = []

func _init() -> void:
	var manifest: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(DIRECTORY + "manifest.json"))
	for item in manifest.assets:
		var image := Image.new()
		if not str(item.file).begins_with("assets/") or str(item.file).contains("..") or FileAccess.get_sha256(DIRECTORY + item.file) != item.sha256 or image.load_png_from_buffer(FileAccess.get_file_as_bytes(DIRECTORY + item.file)) != OK or image.get_size() != Vector2i(item.native_size_px[0], item.native_size_px[1]):
			errors.append("OFF asset validation failed: " + item.id)
			continue
		textures[item.file] = ImageTexture.create_from_image(image)

func place(parent: Control, original_layer: Dictionary, locale: String) -> Node2D:
	var original_id: String = original_layer.id
	var wanted := original_id
	if original_id.begins_with("modal_switch_on_"):
		wanted = original_id.replace("modal_switch_on_", "modal_switch_off_")
	for layer in spec.locales[locale].layers:
		if layer.id != wanted or not textures.has(layer.file):
			continue
		var node := Node2D.new()
		node.name = "PauseOff_" + original_id
		node.position = Layout.reference_transform() * Vector2(layer.top_left_px[0], layer.top_left_px[1])
		node.scale = Layout.reference_transform().get_scale()
		node.z_index = int(layer.z)
		node.set_meta("asset_id", str(layer.file).get_file())
		node.set_meta("original_layer_id", original_id)
		var sprite := Sprite2D.new()
		sprite.name = "SourceOffPixels"
		sprite.centered = false
		sprite.texture = textures[layer.file]
		node.add_child(sprite)
		parent.add_child(node)
		return node
	errors.append("Missing OFF state for " + original_id)
	return null

func label(parent: Control, category: String, locale: String) -> Control:
	for definition in spec.locales[locale].labels:
		if definition.id != category + "_state":
			continue
		var binding := Typography.field("STATIC_PAUSE_OFF_PRIORITY_v1/scene_placement_and_labels.json", definition.id, locale)
		if binding.is_empty():
			errors.append("Missing OFF label typography")
			return null
		var box: Array = definition.rect_px
		var text := DynamicText.new()
		text.name = "OffLabel_" + category
		text.text = definition.text
		text.font = Typography.font(binding.actual_font_binding)
		text.font_size = int(binding.font_px_reference)
		text.minimum_font_size = text.font_size
		text.color = Color(definition.color)
		text.position = Layout.reference_transform() * Vector2(box[0], box[1])
		text.size = Vector2(box[2], box[3])
		text.scale = Layout.reference_transform().get_scale()
		text.z_index = 50
		text.mouse_filter = Control.MOUSE_FILTER_IGNORE
		text.set_meta("font_binding", binding.actual_font_binding)
		parent.add_child(text)
		return text
	return null
